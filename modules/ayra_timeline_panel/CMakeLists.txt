cmake_minimum_required(VERSION 3.22)
project(AyraTimelinePanel LANGUAGES CXX)
add_library(ayra_timeline_panel INTERFACE)
target_include_directories(ayra_timeline_panel INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(ayra_timeline_panel INTERFACE cxx_std_20)
if(PROJECT_IS_TOP_LEVEL)
  include(CTest)
  add_executable(ayra_timeline_panel_tests ../../tests/TimelinePanelTests.cpp)
  target_link_libraries(ayra_timeline_panel_tests PRIVATE ayra_timeline_panel)
  if(MSVC)
    target_compile_options(ayra_timeline_panel_tests PRIVATE /W4 /utf-8)
  else()
    target_compile_options(ayra_timeline_panel_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME timeline_panel_contracts COMMAND ayra_timeline_panel_tests)
endif()
if(PROJECT_IS_TOP_LEVEL)
  add_executable(ayra_panel_fixture_tests ../../tests/PanelFixtureTests.cpp ../../core/LabModel.cpp)
  target_link_libraries(ayra_panel_fixture_tests PRIVATE ayra_timeline_panel)
  if(MSVC)
    target_compile_options(ayra_panel_fixture_tests PRIVATE /W4 /utf-8)
  else()
    target_compile_options(ayra_panel_fixture_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME timeline_fixture_transactions COMMAND ayra_panel_fixture_tests)
endif()
