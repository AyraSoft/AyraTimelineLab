cmake_minimum_required(VERSION 3.22)
project(AyraMediaEditCore VERSION 0.1.0 LANGUAGES CXX)
option(AYRA_MEDIA_EDIT_SANITIZE "Enable ASan and UBSan in the standalone tests" OFF)
add_library(ayra_media_edit INTERFACE)
target_include_directories(ayra_media_edit INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(ayra_media_edit INTERFACE cxx_std_20)
include(CTest)
if(BUILD_TESTING)
  add_executable(ayra_media_edit_tests tests/MediaEditTests.cpp)
  target_link_libraries(ayra_media_edit_tests PRIVATE ayra_media_edit)
  set_target_properties(ayra_media_edit_tests PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(ayra_media_edit_tests PRIVATE /W4 /utf-8)
  else()
    target_compile_options(ayra_media_edit_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(AYRA_MEDIA_EDIT_SANITIZE)
      target_compile_options(ayra_media_edit_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(ayra_media_edit_tests PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME media_edit_contracts COMMAND ayra_media_edit_tests)
endif()
