cmake_minimum_required(VERSION 3.22)
project(AyraMediaDocumentPrototype LANGUAGES CXX)
include(CTest)
option(AYRA_MEDIA_DOCUMENT_SANITIZE "Enable address and undefined-behaviour sanitizers" OFF)
add_library(ayra_media_document_prototype INTERFACE)
target_include_directories(ayra_media_document_prototype INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(ayra_media_document_prototype INTERFACE cxx_std_20)
if(BUILD_TESTING)
    add_executable(ayra_media_document_tests CaptionDocumentTests.cpp)
    target_link_libraries(ayra_media_document_tests PRIVATE ayra_media_document_prototype)
    set_target_properties(ayra_media_document_tests PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(ayra_media_document_tests PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(ayra_media_document_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(AYRA_MEDIA_DOCUMENT_SANITIZE)
            target_compile_options(ayra_media_document_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(ayra_media_document_tests PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME media_document_contracts COMMAND ayra_media_document_tests)
endif()
