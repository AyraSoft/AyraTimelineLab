cmake_minimum_required(VERSION 3.22)
project(AyraMidiPreviewRasterTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${JUCE_SOURCE_DIR}/CMakeLists.txt")
 message(FATAL_ERROR "Set JUCE_SOURCE_DIR to an official JUCE checkout")
endif()
add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
add_executable(ayra_midi_preview_raster_tests Main.cpp)
target_link_libraries(ayra_midi_preview_raster_tests PRIVATE juce::juce_graphics juce::juce_recommended_config_flags)
target_compile_definitions(ayra_midi_preview_raster_tests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
enable_testing()
add_test(NAME midi_preview_pixels COMMAND ayra_midi_preview_raster_tests "${CMAKE_BINARY_DIR}/evidence")
