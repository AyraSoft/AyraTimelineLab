cmake_minimum_required(VERSION 3.22)
project(AyraTimelineIdentityReview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_SOURCE_DIR "" CACHE PATH "Official local JUCE checkout")
set(AYRA_SOURCE_DIR "" CACHE PATH "Optional original Multiverse visual sources")
set(AYRA_MODULES_DIR "" CACHE PATH "Optional ayra_timeline provider parent")
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME visual_adapter_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_visual_adapter.py")
include(../../cmake/VisualSource.cmake)
if(NOT EXISTS "${JUCE_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "JUCE_SOURCE_DIR is required; no network or stub fallback")
endif()
add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
add_executable(ayra_identity_tests IdentityTests.cpp ../../ayra_ui/ayra_ui.cpp ${AYRA_STANDARD_ARTWORK_SOURCE})
target_include_directories(ayra_identity_tests PRIVATE ../../ayra_ui ../../app)
target_link_libraries(ayra_identity_tests PRIVATE ayra_lab_visual_source juce::juce_gui_extra juce::juce_audio_basics juce::juce_recommended_config_flags)
target_compile_definitions(ayra_identity_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
if(APPLE)
  target_link_libraries(ayra_identity_tests PRIVATE "-framework Accelerate")
endif()
add_test(NAME identity_visual_contracts COMMAND ayra_identity_tests "${CMAKE_BINARY_DIR}/identity-evidence")
if(AYRA_MODULES_DIR)
  add_subdirectory(../../modules/ayra_timeline_panel panel)
  add_executable(ayra_identity_input_tests ../NativePanelTests.cpp)
  target_link_libraries(ayra_identity_input_tests PRIVATE ayra_timeline_panel juce::juce_gui_basics juce::juce_recommended_config_flags)
  target_compile_definitions(ayra_identity_input_tests PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  add_test(NAME identity_input_contracts COMMAND ayra_identity_input_tests)
endif()
